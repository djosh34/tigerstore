func TestMemberCPUUsesAllThreadCgroupCountersAndFailsClosed(t *testing.T) {
	var paths []string
	for i := range 3 {
		path := filepath.Join(t.TempDir(), "cpu.stat")
		if err := os.WriteFile(path, []byte(fmt.Sprintf("usage_usec %d\nuser_usec 1\nsystem_usec 2\n", i+10)), 0o600); err != nil {
			t.Fatal(err)
		}
		paths = append(paths, path)
	}
	got, err := memberCPU(strings.Join(paths, ","))
	if err != nil || len(got) != 3 || got[0] != 10 || got[1] != 11 || got[2] != 12 {
		t.Fatalf("counters %v, error %v", got, err)
	}
	if _, err := memberCPU(paths[0]); err == nil {
		t.Fatal("partial member inventory accepted")
	}
	if err := os.WriteFile(paths[1], []byte("user_usec 1\n"), 0o600); err != nil {
		t.Fatal(err)
	}
	if _, err := memberCPU(strings.Join(paths, ",")); err == nil {
		t.Fatal("missing usage_usec accepted as zero")
	}
}

func TestProbeJSONUsesNativeLowercaseCase(t *testing.T) {
	encoded, err := json.Marshal(probe{Case: "overlapping-prev-kv"})
	if err != nil {
		t.Fatal(err)
	}
	var fields map[string]json.RawMessage
	if err := json.Unmarshal(encoded, &fields); err != nil {
		t.Fatal(err)
	}
	if string(fields["case"]) != `"overlapping-prev-kv"` || fields["Case"] != nil {
		t.Fatalf("adapter requires the native lowercase case, got %s", encoded)
	}
}
