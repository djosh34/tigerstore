package wl

import (
	"encoding/json"
	"os"
	"path/filepath"
	"runtime"
	"runtime/pprof"
	"syscall"
	"testing"
	"time"

	"golang.org/x/sys/unix"
)

// Diagnostic-only component isolation, not a rolling test or qualification.
// Scan the immutable failed-run node1 WAL with the existing production scanner.
func TestDiagnosticSavedWALScan(t *testing.T) {
	dir, out := os.Getenv("SAVED_WAL_DIR"), os.Getenv("SCAN_OUT")
	if dir == "" || out == "" {
		t.Skip("scratch diagnostic only")
	}
	for _, mode := range []string{"cold", "warm"} {
		t.Run(mode, func(t *testing.T) {
			if mode == "cold" {
				if err := filepath.WalkDir(dir, func(path string, e os.DirEntry, err error) error {
					if err != nil {
						return err
					}
					if e.IsDir() {
						return nil
					}
					f, err := os.Open(path)
					if err != nil {
						return err
					}
					defer f.Close()
					return unix.Fadvise(int(f.Fd()), 0, 0, unix.FADV_DONTNEED)
				}); err != nil {
					t.Fatal(err)
				}
			}
			runtime.GC()
			profile, err := os.Create(filepath.Join(out, mode+".pprof"))
			if err != nil {
				t.Fatal(err)
			}
			if err := pprof.StartCPUProfile(profile); err != nil {
				t.Fatal(err)
			}
			var before, after syscall.Rusage
			if err := syscall.Getrusage(syscall.RUSAGE_SELF, &before); err != nil {
				t.Fatal(err)
			}
			start := time.Now()
			records := 0
			end, err := scanWAL(sysEnv{}, dir, 0, func(*walRecord) error { records++; return nil })
			elapsed := time.Since(start)
			if err2 := syscall.Getrusage(syscall.RUSAGE_SELF, &after); err2 != nil {
				t.Fatal(err2)
			}
			pprof.StopCPUProfile()
			if err2 := profile.Close(); err2 != nil {
				t.Fatal(err2)
			}
			cpu := func(v syscall.Timeval) float64 { return float64(v.Sec) + float64(v.Usec)/1e6 }
			result := map[string]any{"diagnostic_only": true, "not_a_rolling_reproduction": true, "mode": mode, "source_ref": os.Getenv("SCAN_REF"), "records": records, "end": end, "wall_seconds": elapsed.Seconds(), "user_seconds": cpu(after.Utime) - cpu(before.Utime), "sys_seconds": cpu(after.Stime) - cpu(before.Stime), "max_rss_kib": after.Maxrss}
			if err != nil {
				result["error"] = err.Error()
			}
			b, err2 := json.MarshalIndent(result, "", "  ")
			if err2 != nil {
				t.Fatal(err2)
			}
			if err2 := os.WriteFile(filepath.Join(out, mode+".json"), append(b, '\n'), 0600); err2 != nil {
				t.Fatal(err2)
			}
			t.Log(string(b))
			if err != nil {
				t.Fatal(err)
			}
		})
	}
}
