package wl

import (
	"golang.org/x/sys/unix"
	"os"
	"sync/atomic"
)

// Measurement-only overlay; no counters or branches enter the release binary.
// The identical syscall wrappers are compiled into both immutable refs.
var fetchMeasureReads, fetchMeasureBytes, fetchMeasureNoWait, fetchMeasureNoWaitBytes, fetchMeasureRefused atomic.Uint64
var fetchMeasureRefuse = os.Getenv("FETCH_MEASURE_NOWAIT_REFUSED") == "1"

type fetchMeasureIO struct {
	Reads       uint64 `json:"blocking_reads"`
	Bytes       uint64 `json:"blocking_bytes"`
	NoWait      uint64 `json:"nowait_calls"`
	NoWaitBytes uint64 `json:"nowait_bytes"`
	Refused     uint64 `json:"nowait_refusals"`
}

func fetchMeasureIOSnapshot() fetchMeasureIO {
	return fetchMeasureIO{fetchMeasureReads.Load(), fetchMeasureBytes.Load(), fetchMeasureNoWait.Load(), fetchMeasureNoWaitBytes.Load(), fetchMeasureRefused.Load()}
}
func (a fetchMeasureIO) sub(b fetchMeasureIO) fetchMeasureIO {
	return fetchMeasureIO{a.Reads - b.Reads, a.Bytes - b.Bytes, a.NoWait - b.NoWait, a.NoWaitBytes - b.NoWaitBytes, a.Refused - b.Refused}
}
func fetchMeasurePread(fd int, p []byte, off int64) (int, error) {
	n, err := unix.Pread(fd, p, off)
	fetchMeasureReads.Add(1)
	if n > 0 {
		fetchMeasureBytes.Add(uint64(n))
	}
	return n, err
}
func fetchMeasurePreadNoWait(fd int, p []byte, off int64) (int, error) {
	fetchMeasureNoWait.Add(1)
	if fetchMeasureRefuse {
		fetchMeasureRefused.Add(1)
		return 0, unix.EAGAIN
	}
	n, err := unix.Preadv2(fd, [][]byte{p}, off, unix.RWF_NOWAIT)
	if n > 0 {
		fetchMeasureNoWaitBytes.Add(uint64(n))
	}
	if err != nil {
		fetchMeasureRefused.Add(1)
	}
	return n, err
}
