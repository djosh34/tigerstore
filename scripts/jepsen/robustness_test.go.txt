// This package is copied into the pinned etcd tests module in Actions.
// The upstream generators, recording clients, model and validators are not
// replaced. TestExternalTraffic records a history against a running
// cluster, and TestExternalValidate checks it once the nodes have stopped.

var presets = map[string]traffic.Traffic{
	"etcd-put-delete-lease": traffic.EtcdPutDeleteLease,
	"etcd-put": traffic.EtcdPut,
	"etcd-delete": traffic.EtcdDelete,
	"kubernetes": traffic.Kubernetes,
	"kubernetes-create-delete": traffic.KubernetesCreateDelete,
}

func TestExternalTraffic(t *testing.T) {
	endpoints := strings.Split(os.Getenv("ENDPOINTS"), ",")
	if len(endpoints) != 3 || os.Getenv("GITHUB_ACTIONS") != "true" {
		t.Fatal("requires three fresh endpoints on a GitHub runner")
	}
	tf, ok := presets[os.Getenv("TRAFFIC")]
	if !ok {
		t.Fatal("unknown TRAFFIC preset")
	}
	out := os.Getenv("RESULTS")
	lg := zap.NewExample()
	ctx, cancel := context.WithTimeout(context.Background(), 70*time.Second)
	defer cancel()
	cs := client.NewSet(identity.NewIDProvider(), time.Now())
	defer cs.Close()
	c, err := cs.NewClient(endpoints)
	if err != nil {
		t.Fatal(err)
	}
	// Record revision 1. A prefix on a previously used database is not enough.
	for {
		readCtx, stop := context.WithTimeout(ctx, time.Second)
		r, err := c.Get(readCtx, "key")
		stop()
		if err == nil {
			if r.Header.Revision != 1 || len(r.Kvs) != 0 {
				t.Fatal("cluster is not fresh")
			}
			break
		}
		if ctx.Err() != nil {
			t.Fatal(ctx.Err())
		}
		time.Sleep(100 * time.Millisecond)
	}
	finish := make(chan struct{})
	var wg sync.WaitGroup
	limiter := rate.NewLimiter(50, 50)
	keys := traffic.NewKeyStore(10, "key")
	storage := traffic.NewKubernetesStorage()
	profile := traffic.KeyValueVeryLow
	profile.MaximalQPS = 50
	profile.BurstableQPS = 50
	watch := traffic.WatchDefault
	start := time.Since(cs.BaseTime())
	if err := traffic.SimulateKeyValueTraffic(ctx, &wg, &profile, endpoints, cs, tf, traffic.RunTrafficLoopParam{
		QPSLimiter: limiter, IDs: cs.IdentityProvider(), LeaseIDStorage: identity.NewLeaseIDStorage(),
		NonUniqueRequestConcurrencyLimiter: traffic.NewConcurrencyLimiter(3),
		KeyStore: keys, Storage: storage, Finish: finish,
	}); err != nil {
		t.Fatal(err)
	}
	if err := traffic.SimulateWatchTraffic(ctx, &wg, &watch, endpoints, cs, tf, traffic.RunWatchLoopParam{
		Config: watch, QPSLimiter: limiter, KeyStore: keys, Storage: storage, Finish: finish, Logger: lg,
	}); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(out, "traffic-ready"), []byte("ready\n"), 0600); err != nil {
		t.Fatal(err)
	}
	recoveryErr := awaitRecovery(ctx, out, endpoints)
	if recoveryErr == nil {
		// etcd's suite also runs traffic for 1 s after its failpoint.
		time.Sleep(time.Second)
	}
	close(finish)
	wg.Wait()
	end := time.Since(cs.BaseTime())
	putCtx, stop := context.WithTimeout(ctx, 10*time.Second)
	_, putErr := c.Put(putCtx, "tombstone", "true")
	stop()
	reports := cs.Reports()
	// Save typed operations before checking, including on a timeout or failure.
	r, err := report.NewTestReport(lg, filepath.Join(out, "reports"), nil, nil)
	if err != nil {
		t.Fatal(err)
	}
	r.SetClientReports(reports)
	if err := r.SaveEtcdData(); err != nil {
		t.Fatal(err)
	}
	if recoveryErr != nil {
		t.Fatal(recoveryErr)
	}
	if putErr != nil {
		t.Fatalf("terminal put failed: %v", putErr)
	}
	stats := traffic.CalculateStats(reports, start, end)
	watches := traffic.CalculateWatchStats(reports, start, end)
	writeJSON(t, filepath.Join(out, "traffic.json"), map[string]any{
		"traffic": os.Getenv("TRAFFIC"), "fault": filepath.Base(out), "stats": stats, "watch_traffic": watches,
	})
	if stats.Successes < 200 || watches.Events == 0 {
		t.Error("insufficient successful KV operations or no watch events")
	}
}

// TestExternalValidate checks the saved history with etcd's validators. As
// in etcd's suite, the requests the stopped cluster persisted resolve
// writes of unknown outcome, and the watch and historical-read checks
// replay them. WAL_DIRS names etcd data directories: the nodes' own for
// etcd, and the one cmd/etcdwal wrote from the WALs for tigerstore.
func TestExternalValidate(t *testing.T) {
	tf, ok := presets[os.Getenv("TRAFFIC")]
	if !ok {
		t.Fatal("unknown TRAFFIC preset")
	}
	out := os.Getenv("RESULTS")
	lg := zap.NewExample()
	reports, err := report.LoadClientReports(filepath.Join(out, "reports"))
	if err != nil {
		t.Fatal(err)
	}
	persisted, err := report.PersistedRequests(lg, strings.Split(os.Getenv("WAL_DIRS"), ","))
	if err != nil {
		t.Fatal(err)
	}
	checkStart := time.Now()
	result := validate.ValidateAndReturnVisualize(lg, validate.Config{
		ExpectRevisionUnique: tf.ExpectUniqueRevision(),
	}, reports, persisted, 5*time.Minute)
	writeJSON(t, filepath.Join(out, "summary.json"), map[string]any{
		"traffic": os.Getenv("TRAFFIC"), "fault": filepath.Base(out),
		"persisted_requests": len(persisted), "check_seconds": time.Since(checkStart).Seconds(),
		"assumptions": result.Assumptions, "linearization": result.Linearization.Result,
		"watch": result.Watch, "historical_reads": result.Serializable,
	})
	// Result.Error accepts Unknown, so each status must be Success.
	for name, r := range map[string]validate.Result{
		"assumptions": result.Assumptions, "linearization": result.Linearization.Result,
		"watch": result.Watch, "historical reads": result.Serializable,
	} {
		if r.Status != validate.Success {
			t.Errorf("%s validation: %v", name, r)
		}
	}
}

func writeJSON(t *testing.T, path string, v any) {
	data, err := json.MarshalIndent(v, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, data, 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("%s", data)
}

// awaitRecovery waits until the runner has finished the fault and every
// endpoint answers a linearizable read. Its client is not recorded.
func awaitRecovery(ctx context.Context, out string, endpoints []string) error {
	for {
		if _, err := os.Stat(filepath.Join(out, "fault-done")); err == nil {
			break
		}
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(100 * time.Millisecond):
		}
	}
	for _, endpoint := range endpoints {
		cli, err := clientv3.New(clientv3.Config{Endpoints: []string{endpoint}, DialTimeout: time.Second})
		if err != nil {
			return err
		}
		for {
			getCtx, stop := context.WithTimeout(ctx, time.Second)
			_, err = cli.Get(getCtx, "key")
			stop()
			if err == nil || ctx.Err() != nil {
				break
			}
			time.Sleep(100 * time.Millisecond)
		}
		if closeErr := cli.Close(); closeErr != nil {
			return closeErr
		}
		if err != nil {
			return fmt.Errorf("%s did not answer after the fault: %w", endpoint, err)
		}
	}
	return nil
}
